```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
```

```python
def f(x, y):
    return np.sin(x) ** 10 + np.cos(10 + y*x) * np.cos(x)
```

```python
x = np.linspace(0, 5, 50)
y = np.linspace(0, 5, 40)

X, Y = np.meshgrid(x, y)

Z = f(X, Y)
```

```python
plt.contour(X, Y, Z, colors='black')
```

```python
plt.contour(X, Y, Z, 20, cmap='RdGy')
```

```python
plt.contourf(X, Y, Z, 20, cmap='RdGy')
plt.colorbar()
```

```python
plt.imshow(Z, extent=[0, 5, 0, 5], origin='lower', cmap='RdGy')
plt.colorbar()
plt.axis('equal')
```

```python
countours = plt.contour(X, Y, Z, 3, colors='black')

plt.clabel(countours, inline=True, fontsize=8) # підпис ліній, inline = True значить, що наша лінія розривається у місці, де стоїть значення (естетика)
```
