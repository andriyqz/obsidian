```python
df6 = pd.DataFrame({'name': ['Peter', 'Paul', 'Mary'],
                    'food': ['fish', 'beans', 'bread']})

df7 = pd.DataFrame({'name': ['Mary', 'Joseph'],
                    'drinks': ['wine', 'beer']})

  

print(pd.merge(df6, df7)) # how='inner' by default
print(pd.merge(df6, df7, how='outer')) # усі колонки
print(pd.merge(df6, df7, how='left')) # усі стовпці лівого датафрейму повинні бути, якщо у правому датафреймові немає відповідності - заповнюємо NA значенням
                                      # для right аналогічно
```

```python
pop = pd.read_csv('state-population.csv')
areas = pd.read_csv('state-areas.csv')
abbrevs = pd.read_csv('state-abbrevs.csv')

print(pop.head())
print(areas.head())
print(abbrevs.head())
```

```python
merged = pd.merge(pop, abbrevs, how='outer', left_on='state/region', right_on='abbreviation')
merged = merged.drop('abbreviation', axis=1)

merged
```
